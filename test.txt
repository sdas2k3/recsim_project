# Import necessary libraries
import pandas as pd
import numpy as np
from sklearn.model_selection import KFold, cross_val_score
from sklearn.preprocessing import OneHotEncoder, LabelEncoder
from sklearn.compose import ColumnTransformer
from sklearn.pipeline import Pipeline
from sklearn.ensemble import RandomForestClassifier
from sklearn.neural_network import MLPClassifier
from xgboost import XGBClassifier
from sklearn.metrics import accuracy_score, classification_report, confusion_matrix
import warnings
warnings.filterwarnings('ignore')

# Load your dataset (replace 'your_dataset.csv' with actual file path)
# df = pd.read_csv('your_dataset.csv')

# For demonstration, creating sample data
# Remove this section and uncomment the line above when using your actual data
np.random.seed(42)
sample_data = {
    'modelcode': np.random.choice(['A001', 'B002', 'C003', 'D004', 'E005'], 1000),
    'bir_date': pd.date_range('2020-01-01', periods=1000, freq='D').strftime('%Y%m%d'),
    'rec_sym_all': np.random.choice(['Y', 'N'], 1000),
    'Mid_category_cd': np.random.choice(['CAT1', 'CAT2', 'CAT3', 'CAT4'], 1000)
}
df = pd.DataFrame(sample_data)

print("Dataset shape:", df.shape)
print("\nDataset info:")
print(df.info())
print("\nFirst few rows:")
print(df.head())

# Data preprocessing functions
def extract_birth_month(date_str):
    """Extract month from bir_date (YYYYMMDD format)"""
    try:
        return date_str[4:6]  # Extract MM part
    except:
        return '01'  # Default to January if error

# Create birth month feature
df['birth_month'] = df['bir_date'].apply(extract_birth_month)

# Prepare features and target
X = df[['modelcode', 'birth_month', 'rec_sym_all']]
y = df['Mid_category_cd']

print("\nFeature columns:", X.columns.tolist())
print("Target distribution:")
print(y.value_counts())

# Create preprocessing pipeline
# One-hot encode modelcode and birth_month, rec_sym_all
preprocessor = ColumnTransformer(
    transformers=[
        ('cat', OneHotEncoder(drop='first', sparse_output=False), ['modelcode', 'birth_month', 'rec_sym_all'])
    ],
    remainder='passthrough'
)

# Initialize models
models = {
    'XGBoost': XGBClassifier(random_state=42, n_estimators=100),
    'Random Forest': RandomForestClassifier(n_estimators=100, random_state=42),
    'MLP': MLPClassifier(hidden_layer_sizes=(100, 50), max_iter=500, random_state=42)
}

# Initialize KFold
kf = KFold(n_splits=10, shuffle=True, random_state=42)

# Store results
results = {}

# Train and evaluate models
print("\n" + "="*60)
print("TRAINING AND EVALUATION RESULTS")
print("="*60)

for model_name, model in models.items():
    print(f"\nTraining {model_name}...")
    
    # Create pipeline with preprocessing and model
    pipeline = Pipeline([
        ('preprocessor', preprocessor),
        ('classifier', model)
    ])
    
    # Perform cross-validation
    cv_scores = cross_val_score(pipeline, X, y, cv=kf, scoring='accuracy')
    
    # Store results
    results[model_name] = {
        'cv_scores': cv_scores,
        'mean_cv_score': cv_scores.mean(),
        'std_cv_score': cv_scores.std(),
        'pipeline': pipeline
    }
    
    print(f"{model_name} - CV Accuracy: {cv_scores.mean():.4f} (+/- {cv_scores.std() * 2:.4f})")

# Detailed evaluation with train/test split for best model
print("\n" + "="*60)
print("DETAILED EVALUATION ON TEST SETS")
print("="*60)

from sklearn.model_selection import train_test_split

# Find best model based on CV scores
best_model_name = max(results.keys(), key=lambda x: results[x]['mean_cv_score'])
print(f"\nBest model based on CV: {best_model_name}")

# Detailed KFold evaluation for all models
all_results = {}

for model_name, model_info in results.items():
    print(f"\n--- {model_name} Detailed Evaluation ---")
    
    fold_scores = []
    fold_predictions = []
    fold_true_values = []
    
    for fold, (train_idx, test_idx) in enumerate(kf.split(X), 1):
        # Split data
        X_train_fold, X_test_fold = X.iloc[train_idx], X.iloc[test_idx]
        y_train_fold, y_test_fold = y.iloc[train_idx], y.iloc[test_idx]
        
        # Train model
        model_info['pipeline'].fit(X_train_fold, y_train_fold)
        
        # Predict
        y_pred_fold = model_info['pipeline'].predict(X_test_fold)
        
        # Calculate accuracy
        fold_accuracy = accuracy_score(y_test_fold, y_pred_fold)
        fold_scores.append(fold_accuracy)
        
        # Store predictions for final analysis
        fold_predictions.extend(y_pred_fold)
        fold_true_values.extend(y_test_fold)
        
        print(f"Fold {fold}: Accuracy = {fold_accuracy:.4f}")
    
    # Overall results for this model
    mean_accuracy = np.mean(fold_scores)
    std_accuracy = np.std(fold_scores)
    
    all_results[model_name] = {
        'fold_scores': fold_scores,
        'mean_accuracy': mean_accuracy,
        'std_accuracy': std_accuracy,
        'predictions': fold_predictions,
        'true_values': fold_true_values
    }
    
    print(f"Overall {model_name}: {mean_accuracy:.4f} (+/- {std_accuracy * 2:.4f})")

# Summary of all models
print("\n" + "="*60)
print("FINAL MODEL COMPARISON")
print("="*60)

comparison_df = pd.DataFrame({
    'Model': list(all_results.keys()),
    'Mean Accuracy': [all_results[model]['mean_accuracy'] for model in all_results.keys()],
    'Std Accuracy': [all_results[model]['std_accuracy'] for model in all_results.keys()]
})

comparison_df = comparison_df.sort_values('Mean Accuracy', ascending=False)
print(comparison_df)

# Display detailed results for best model
best_model = comparison_df.iloc[0]['Model']
print(f"\n--- Best Model: {best_model} ---")
print(f"Mean Accuracy: {comparison_df.iloc[0]['Mean Accuracy']:.4f}")
print(f"Std Accuracy: {comparison_df.iloc[0]['Std Accuracy']:.4f}")

# Feature importance for tree-based models
if best_model in ['XGBoost', 'Random Forest']:
    print(f"\n--- Feature Importance for {best_model} ---")
    
    # Get the best model pipeline
    best_pipeline = results[best_model]['pipeline']
    
    # Fit on entire dataset to get feature importance
    best_pipeline.fit(X, y)
    
    if best_model == 'XGBoost':
        feature_importance = best_pipeline.named_steps['classifier'].feature_importances_
    else:  # Random Forest
        feature_importance = best_pipeline.named_steps['classifier'].feature_importances_
    
    # Get feature names after preprocessing
    feature_names = best_pipeline.named_steps['preprocessor'].get_feature_names_out()
    
    # Create feature importance dataframe
    importance_df = pd.DataFrame({
        'feature': feature_names,
        'importance': feature_importance
    }).sort_values('importance', ascending=False)
    
    print("Top 10 Most Important Features:")
    print(importance_df.head(10))

# Confusion Matrix for best model
print(f"\n--- Confusion Matrix for {best_model} ---")
best_predictions = all_results[best_model]['predictions']
best_true_values = all_results[best_model]['true_values']

from sklearn.metrics import ConfusionMatrixDisplay
import matplotlib.pyplot as plt

plt.figure(figsize=(8, 6))
cm = confusion_matrix(best_true_values, best_predictions)
disp = ConfusionMatrixDisplay(confusion_matrix=cm, display_labels=np.unique(y))
disp.plot(cmap='Blues')
plt.title(f'Confusion Matrix - {best_model}')
plt.show()

# Save results to CSV
print("\nSaving results to CSV...")
comparison_df.to_csv('model_comparison_results.csv', index=False)
print("Results saved to 'model_comparison_results.csv'")

# Optional: Save the best model
import joblib
best_pipeline = results[best_model]['pipeline']
best_pipeline.fit(X, y)  # Fit on entire dataset
joblib.dump(best_pipeline, f'best_model_{best_model.lower().replace(" ", "_")}.pkl')
print(f"Best model saved as 'best_model_{best_model.lower().replace(' ', '_')}.pkl'")

print("\n" + "="*60)
print("ANALYSIS COMPLETE")
print("="*60)
